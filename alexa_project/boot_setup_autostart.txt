#ini edit:

[Unit]
Description=AlexaLite Assistant
After=network.target sound.target

[Service]
ExecStart=/usr/bin/python3 /home/alexa/main.py
WorkingDirectory=/home/alexa

User=root

[Install]
WantedBy=multi-user.target


#Enable:
